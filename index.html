@RestController
@RequestMapping("/api/v1/jobs")
@CrossOrigin(origins = "*")
public class JobPortalController {

  @Autowired
  private JobService jobService;

  @PostMapping("/post")
  public ResponseEntity<Job> postJob(
      @Valid @RequestBody JobDto dto) {
    
    // Business validation & DB persistence
    Job saved = jobService.createJobListing(dto);
    return new ResponseEntity<>(saved, HttpStatus.CREATED);
  }

  @GetMapping("/search")
  public List<Job> searchJobs(@RequestParam String keyword) {
    return jobService.filterJobs(keyword);
  }
}
